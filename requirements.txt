# --- core ------------------------------------------------------------------
# numpy must stay below 2.0: the gym/sconegym generation this project uses
# predates the numpy 2 ABI change.
numpy>=1.21,<2.0
pyyaml>=5.4

# --- RL stack --------------------------------------------------------------
# gym is deliberately unpinned here because the working environment uses a
# pre-0.22 release (the old `registry.make(id, **kwargs)` API). Upgrading gym
# WILL break sconegym. Pin it to whatever your working machine reports -- see
# the freeze recipe below -- rather than letting pip pick the newest.
gym<0.22
torch>=1.12
tensorboard

# deprl provides custom_agents, custom_mpo_torch, custom_replay_buffers and
# custom_trainer, all referenced from the stage configs.
deprl

# --- plots and notebook ----------------------------------------------------
matplotlib>=3.5,<4
jupyterlab>=3.4
ipykernel>=6.0
nbformat>=5.0

# --- tests and tooling -----------------------------------------------------
pytest>=7.0
# psutil is optional. Without it run_stages.py still limits BLAS/OpenMP thread
# counts, but the process CPU affinity mask (the hard cap) is not applied.
psutil>=5.9
